```yaml
id: R3-L03
inherits: ../../ppt_design_mechanism_library/global_defaults.yaml
name: dual_system_convergence
industry_specific: false
relation: distinct_system_A + distinct_system_B -> one_shared_outcome

match_when:
  - 恰好有两套性质不同但互补的系统、能力或路径
  - 每侧至少有2项机制证据
  - 两侧共同强化同一结果、目标或定位
  - 左右证据类型、数量或视觉形式可以不同

reject_when:
  - 两侧没有共同结果，只是普通并列
  - 两侧互相替代或竞争
  - A严格先于B发生
  - 存在3个以上同等重要支柱

slots:
  required:
    system_A: 第一套机制
    system_B: 第二套机制
    shared_outcome: 共同结果
    evidence_A: 2–5项
    evidence_B: 2–5项
  optional:
    joint_metrics: 0–3项共同结果指标
  bindings:
    - evidence_A只能支持system_A或shared_outcome
    - evidence_B只能支持system_B或shared_outcome
    - joint_metric必须同时能回指A与B，否则放回单侧

variants:
  V1:
    layout: 左右系统 + 中央汇聚点
    use_when: 两侧证据量接近
  V2:
    layout: 左右系统 + 底部共同结果栏
    use_when: 共同结果包含1–3个可量化指标
  V3:
    layout: 主系统55%–62% + 辅助系统31%–38% + 汇聚点
    use_when: 两侧都必要但证据量明显不同
  V4:
    layout: 两条路径由左右向共同目标汇合
    use_when: 强调协同方向但不存在先后顺序

layout_dna:
  - 左右各承载一套完整机制，中间或底部必须出现明确共同结果
  - 对称的是论证职责，不要求内部组件、媒介和宽度相同
  - 左右默认视觉权重55:40，可按证据量调整
  - 汇聚点只承载共同结果，不增加第三套主题
  - 共用指标只显示一次
  - 最多两个大容器，禁止镜像复制同构小卡片

content_limits:
  systems: 2

  joint_metrics: 0–3

assets:
```
